39template<typename T>
40auto approx_equal(T a, T b, T rel_tol = T(100) * epsilon<T>(), T abs_tol = epsilon<T>()) noexcept -> bool
41{
42 static_assert(std::is_floating_point_v<T>, "approx_equal requires floating-point type");
43
44 const T diff = std::abs(a - b);
45
46 // Absolute tolerance check (important for values near zero)
47 if (diff <= abs_tol)
48 {
49 return true;
50 }
51
52 // Relative tolerance check
53 const T max_abs = std::max(std::abs(a), std::abs(b));
54 return diff <= rel_tol * max_abs;
55}

64template<typename T, typename Iterator>
65auto kahan_sum(Iterator begin, Iterator end) noexcept -> T
66{
67 static_assert(std::is_floating_point_v<T>, "kahan_sum requires floating-point type");
68
69 T sum = T(0);
70 T c = T(0); // Compensation for lost low-order bits
71
72 for (Iterator it = begin; it != end; ++it)
73 {
74 T y = *it - c;
75 T t = sum + y;
76 c = (t - sum) - y;
77 sum = t;
78 }
79
80 return sum;
81}

88template<typename T>
89auto hypot(T a, T b) noexcept -> T
90{
91 static_assert(std::is_floating_point_v<T>, "hypot requires floating-point type");
92
93 T abs_a = std::abs(a);
94 T abs_b = std::abs(b);
95
96 if (abs_a == T(0))
97 {
98 return abs_b;
99 }
100 if (abs_b == T(0))
101 {
102 return abs_a;
103 }
104
105 // Scale to avoid overflow
106 T max_val = std::max(abs_a, abs_b);
107 T min_val = std::min(abs_a, abs_b);
108
109 T ratio = min_val / max_val;
110 return max_val * std::sqrt(T(1) + ratio * ratio);
111}

120template<typename T>
121auto hypot3(T a, T b, T c) noexcept -> T
122{
123 static_assert(std::is_floating_point_v<T>, "hypot3 requires floating-point type");
124
125 // Use absolute values
126 a = std::abs(a);
127 b = std::abs(b);
128 c = std::abs(c);
129
130 // Find the maximum to use for scaling (avoid initializer list for portability)
131 T max_val = std::max(std::max(a, b), c);
132
133 if (max_val == T(0) || !std::isfinite(max_val))
134 {
135 return max_val;
136 }
137
138 // Check for extreme values that would overflow even after scaling
139 // If max_val > sqrt(T_max) / 2, the result would overflow
140 const T overflow_threshold = std::sqrt(std::numeric_limits<T>::max()) / T(2);
141 if (max_val > overflow_threshold)
142 {
143 return std::numeric_limits<T>::infinity();
144 }
145
146 // Scale by max to prevent overflow/underflow
147 a /= max_val;
148 b /= max_val;
149 c /= max_val;
150
151 return max_val * std::sqrt(a * a + b * b + c * c);
152}

210template<typename T>
211auto condition_number_2x2(T a, T b, T c, T d) noexcept -> T
212{
213 // Handle trivial case (zero matrix)
214 T max_elem = std::max(std::max(std::abs(a), std::abs(b)), std::max(std::abs(c), std::abs(d)));
215 if (max_elem == T(0))
216 {
217 return T(1); // Zero matrix has condition number 1 by convention
218 }
219
220 // Preserve originals for robust checks
221 T a0 = a;
222 T b0 = b;
223 T c0 = c;
224 T d0 = d;
225
226 // Quick check using the raw determinant (before scaling) to detect exact/near-singularity
227 // Use a configurable higher-precision intermediate type (`hi_t`) so embedded
228 // builds can avoid pulling in long-double software helpers.
229 using high_t = microla::detail::hi_t<T>;
230
231 high_t raw_det_hi =
232 static_cast<high_t>(a0) * static_cast<high_t>(d0) - static_cast<high_t>(b0) * static_cast<high_t>(c0);
233
234 // Conservative tolerance scaled to the magnitude of the matrix. Use a larger
235 // multiplier to be robust under aggressive floating-point optimizations.
236 high_t det_tol_hi = static_cast<high_t>(epsilon<T>()) * static_cast<high_t>(max_elem) *
237 static_cast<high_t>(max_elem) * static_cast<high_t>(1e6);
238
239 if (std::abs(raw_det_hi) <= det_tol_hi)
240 {
241 return std::numeric_limits<T>::infinity();
242 }
243
244 // Extra cross-product check (a*d ?= b*c) using higher-precision intermediate
245 // to catch near-proportional rows/cols.
246 high_t cross_lhs = static_cast<high_t>(a0) * static_cast<high_t>(d0);
247 high_t cross_rhs = static_cast<high_t>(b0) * static_cast<high_t>(c0);
248 if (std::abs(cross_lhs - cross_rhs) <= det_tol_hi)
249 {
250 return std::numeric_limits<T>::infinity();
251 }
252
253 // Scale matrix to reduce sensitivity to magnitude
254 a /= max_elem;
255 b /= max_elem;
256 c /= max_elem;
257 d /= max_elem;
258
259 // Calculate A^T A for [[a,b],[c,d]]: A^T A = [[a²+c², ab+cd], [ab+cd, b²+d²]]
260 T aa = a * a + c * c;
261 T bb = b * b + d * d;
262 T ab = a * b + c * d;
263
264 // Eigenvalues of A^T A: λ = (trace ± sqrt(trace² - 4*det)) / 2
265 T trace = aa + bb;
266 T det_ata = aa * bb - ab * ab;
267
268 // Robust tolerance scaled to problem size
269 T tol = epsilon<T>() * std::max(trace * trace, T(1));
270 if (det_ata <= tol || det_ata <= T(0))
271 {
272 return std::numeric_limits<T>::infinity();
273 }
274
275 T discriminant = trace * trace - T(4) * det_ata;
276 // Protect against numerical noise making discriminant negative
277 if (discriminant < T(0))
278 {
279 discriminant = T(0);
280 }
281
282 T sqrt_d = std::sqrt(discriminant);
283 T lambda_max = (trace + sqrt_d) / T(2);
284 T lambda_min = (trace - sqrt_d) / T(2);
285
286 // Check if minimum eigenvalue is too small
287 if (lambda_min <= tol || lambda_min <= T(0))
288 {
289 return std::numeric_limits<T>::infinity();
290 }
291
292 // Condition number κ = σ_max / σ_min = sqrt(λ_max / λ_min)
293 T ratio = lambda_max / lambda_min;
294 return std::sqrt(ratio);
295}

303template<typename T>
304auto stable_dot_product(const T* a, const T* b, std::size_t n) noexcept -> T
305{
306 static_assert(std::is_floating_point_v<T>, "stable_dot_product requires floating-point type");
307
308 T sum = T(0);
309 T c = T(0); // Compensation
310
311 for (std::size_t i = 0; i < n; ++i)
312 {
313 T product = a[i] * b[i];
314 T y = product - c;
315 T t = sum + y;
316 c = (t - sum) - y;
317 sum = t;
318 }
319
320 return sum;
321}

329template<typename T>
330auto horner_eval(const T* coeffs, std::size_t n, T x) noexcept -> T
331{
332 static_assert(std::is_floating_point_v<T>, "horner_eval requires floating-point type");
333
334 if (n == 0)
335 {
336 return T(0);
337 }
338
339 T result = coeffs[n - 1];
340 for (int i = static_cast<int>(n) - 2; i >= 0; --i)
341 {
342 result = result * x + coeffs[i];
343 }
344
345 return result;
346}

378template<typename T>
379auto ulp_distance(T a, T b) noexcept -> std::int64_t
380{
381 static_assert(std::is_floating_point_v<T>, "ulp_distance requires floating-point type");
382
383 // For simplicity, use relative difference scaled by epsilon
384 if (a == b)
385 {
386 return 0;
387 }
388
389 T diff = std::abs(a - b);
390 T eps = epsilon<T>();
391 T scale = std::max(std::abs(a), std::abs(b));
392
393 return static_cast<std::int64_t>(diff / (scale * eps));
394}
