44template<typename T>
45auto approx_equal(T a, T b, T rel_tol = T(100) * epsilon<T>(), T abs_tol = epsilon<T>()) noexcept -> bool
46{
47 static_assert(std::is_floating_point_v<T>, "approx_equal requires floating-point type");
48
49 if (a == b)
50 {
51 return true;
52 }
53 if (!std::isfinite(a) || !std::isfinite(b) || !std::isfinite(rel_tol) || !std::isfinite(abs_tol) ||
54 rel_tol < T(0) || abs_tol < T(0))
55 {
56 return false;
57 }
58 const T diff = std::abs(a - b);
59
60 // Absolute tolerance check (important for values near zero)
61 if (diff <= abs_tol)
62 {
63 return true;
64 }
65
66 // Relative tolerance check
67 const T max_abs = std::max(std::abs(a), std::abs(b));
68 return diff <= rel_tol * max_abs;
69}

78template<typename T, typename Iterator>
79auto kahan_sum(Iterator begin, Iterator end) noexcept -> T
80{
81 static_assert(std::is_floating_point_v<T>, "kahan_sum requires floating-point type");
82
83 T sum = T(0);
84 T c = T(0); // Compensation for lost low-order bits
85
86 for (Iterator it = begin; it != end; ++it)
87 {
88 T y = *it - c;
89 T t = sum + y;
90 c = (t - sum) - y;
91 sum = t;
92 }
93
94 return sum;
95}

189template<typename T>
190auto condition_number_2x2(T a, T b, T c, T d) noexcept -> T
191{
192 T max_elem = std::max(std::max(std::abs(a), std::abs(b)), std::max(std::abs(c), std::abs(d)));
193 if (max_elem == T(0) || !std::isfinite(a) || !std::isfinite(b) || !std::isfinite(c) || !std::isfinite(d))
194 {
195 return std::numeric_limits<T>::infinity();
196 }
197 using high_t = microla::detail::hi_t<T>;
198 const high_t left_top = static_cast<high_t>(a) / max_elem;
199 const high_t right_top = static_cast<high_t>(b) / max_elem;
200 const high_t left_bottom = static_cast<high_t>(c) / max_elem;
201 const high_t right_bottom = static_cast<high_t>(d) / max_elem;
202 const high_t determinant = std::abs(left_top * right_bottom - right_top * left_bottom);
203 if (determinant == static_cast<high_t>(0))
204 {
205 return std::numeric_limits<T>::infinity();
206 }
207 const high_t aa = left_top * left_top + left_bottom * left_bottom;
208 const high_t bb = right_top * right_top + right_bottom * right_bottom;
209 const high_t ab = left_top * right_top + left_bottom * right_bottom;
210 const high_t largest = (aa + bb + std::hypot(aa - bb, static_cast<high_t>(2) * ab)) / static_cast<high_t>(2);
211 return static_cast<T>(largest / determinant);
212}

220template<typename T>
221auto stable_dot_product(const T* a, const T* b, std::size_t n) noexcept -> T
222{
223 static_assert(std::is_floating_point_v<T>, "stable_dot_product requires floating-point type");
224
225 T sum = T(0);
226 T c = T(0); // Compensation
227
228 for (std::size_t i = 0; i < n; ++i)
229 {
230 T product = a[i] * b[i];
231 T y = product - c;
232 T t = sum + y;
233 c = (t - sum) - y;
234 sum = t;
235 }
236
237 return sum;
238}

246template<typename T>
247auto horner_eval(const T* coeffs, std::size_t n, T x) noexcept -> T
248{
249 static_assert(std::is_floating_point_v<T>, "horner_eval requires floating-point type");
250
251 if (n == 0)
252 {
253 return T(0);
254 }
255
256 T result = coeffs[n - 1];
257 for (int i = static_cast<int>(n) - 2; i >= 0; --i)
258 {
259 result = result * x + coeffs[i];
260 }
261
262 return result;
263}

304template<typename T>
305auto ulp_distance(T a, T b) noexcept -> std::int64_t
306{
307 static_assert((std::is_same_v<T, float> || std::is_same_v<T, double>) && std::numeric_limits<T>::is_iec559,
308 "ulp_distance requires IEEE float or double");
309 if (std::isnan(a) || std::isnan(b))
310 {
311 return std::numeric_limits<std::int64_t>::max();
312 }
313 if (a == b)
314 {
315 return 0;
316 }
317
318 using Bits = std::conditional_t<std::is_same_v<T, float>, std::uint32_t, std::uint64_t>;
319 constexpr Bits sign = Bits(1) << (std::numeric_limits<Bits>::digits - 1);
320 const auto ordered = [](T value) -> Bits
321 {
322 const Bits bits = MICROLA_BIT_CAST(Bits, value);
323 return (bits & sign) != Bits(0) ? sign - (bits & ~sign) : sign + bits;
324 };
325 const Bits left = ordered(a);
326 const Bits right = ordered(b);
327 const Bits distance = left > right ? left - right : right - left;
328 const auto limit = static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max());
329 return static_cast<std::int64_t>(std::min(static_cast<std::uint64_t>(distance), limit));
330}
