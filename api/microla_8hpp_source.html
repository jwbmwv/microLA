112template<typename T>
113constexpr auto angle_remainder(T angle) noexcept -> T
114{
115 if constexpr (std::is_integral_v<T>)
116 {
117 return angle % constants::two_pi<T>();
118 }
119 if (!(angle <= std::numeric_limits<T>::max() && angle >= -std::numeric_limits<T>::max()))
120 {
121 return std::numeric_limits<T>::quiet_NaN();
122 }
123 if (!std::is_constant_evaluated())
124 {
125 return std::fmod(angle, constants::two_pi<T>());
126 }
127 const bool negative = angle < T(0);
128 T remainder = negative ? -angle : angle;
129 T divisor = constants::two_pi<T>();
130 for (int step = 0; step < std::numeric_limits<T>::max_exponent && divisor <= remainder / T(2); ++step)
131 {
132 divisor *= T(2);
133 }
134 for (int step = 0; step <= std::numeric_limits<T>::max_exponent && divisor >= constants::two_pi<T>(); ++step)
135 {
136 if (remainder >= divisor)
137 {
138 remainder -= divisor;
139 }
140 divisor /= T(2);
141 }
142 return negative ? -remainder : remainder;
143}

150template<typename T>
151[[nodiscard]] constexpr auto wrap_pi(T angle) noexcept -> T
152{
153 angle = detail::angle_remainder(angle);
154 if (angle > constants::pi<T>())
155 {
156 angle -= constants::two_pi<T>();
157 }
158 if (angle < -constants::pi<T>())
159 {
160 angle += constants::two_pi<T>();
161 }
162 return angle;
163}

169template<typename T>
170[[nodiscard]] constexpr auto wrap_two_pi(T angle) noexcept -> T
171{
172 angle = detail::angle_remainder(angle);
173 if (angle < T(0))
174 {
175 angle += constants::two_pi<T>();
176 }
177 if (angle >= constants::two_pi<T>())
178 {
179 return T(0);
180 }
181 return angle;
182}
