101template<typename T>
102[[nodiscard]] constexpr auto wrap_pi(T angle) noexcept -> T
103{
104 // Normalize to [-pi, pi]
105 while (angle > constants::pi<T>())
106 {
107 angle -= constants::two_pi<T>();
108 }
109 while (angle < -constants::pi<T>())
110 {
111 angle += constants::two_pi<T>();
112 }
113 return angle;
114}

119template<typename T>
120[[nodiscard]] constexpr auto wrap_two_pi(T angle) noexcept -> T
121{
122 // Normalize to [0, 2*pi]
123 while (angle < T(0))
124 {
125 angle += constants::two_pi<T>();
126 }
127 while (angle >= constants::two_pi<T>())
128 {
129 angle -= constants::two_pi<T>();
130 }
131 return angle;
