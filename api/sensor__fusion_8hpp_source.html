144template<typename T>
145constexpr auto timestamp_needs_reset(T timestamp_s) noexcept -> bool
146{
147 static_assert(std::is_floating_point<T>::value, "timestamp_needs_reset requires a floating-point timestamp type");
148 // 2^24 ms ≈ 16 777 s ≈ 4.66 hours — the last timestamp a 32-bit float can
149 // represent with exact 1 ms resolution (24 mantissa bits allow exact integer
150 // millisecond values up to 2^24). For double the threshold is 2^53 ms which
151 // is cosmological, so the check is meaningful only for float; for double we
152 // always return false.
153 if constexpr (sizeof(T) == sizeof(float))
154 {
155 constexpr T k_float_precision_cliff_s = static_cast<T>(1 << 24) / static_cast<T>(1000); // 16 777.216 s
156 return timestamp_s >= k_float_precision_cliff_s;
157 }
158 return false;
159}

726template<typename T>
727inline auto wrap_pi(T angle) noexcept -> T
728{
729 const T pi = constants::pi<T>();
730 const T two_pi = constants::two_pi<T>();
731 // Use fmod for O(1) wrapping regardless of the magnitude of `angle`.
732 // The while-loop alternative is O(|angle|/pi) and degrades for large inputs.
733 angle = std::fmod(angle, two_pi);
734 if (angle > pi)
735 {
736 angle -= two_pi;
737 }
738 else if (angle < -pi)
739 {
740 angle += two_pi;
741 }
742 return angle;
743}

843template<typename T>
844inline auto twist_quaternion_about_axis(const Quaternion<T>& q, const Vec<T, 3>& axis) noexcept -> Quaternion<T>
845{
846 const Vec<T, 3> unit_axis = normalized_or(axis, Vec<T, 3>(T(0), T(0), T(1)));
847 const Vec<T, 3> projected = unit_axis * q.vec().dot(unit_axis);
848 Quaternion<T> twist(q.w(), projected[0], projected[1], projected[2]);
849 if (!(twist.norm_squared() > std::numeric_limits<T>::epsilon()))
850 {
851 return Quaternion<T>::identity();
852 }
853 return twist.normalized();
854}

857template<typename T>
858inline auto twist_angle_about_axis(const Quaternion<T>& q, const Vec<T, 3>& axis) noexcept -> T
859{
860 const Vec<T, 3> unit_axis = normalized_or(axis, Vec<T, 3>(T(0), T(0), T(1)));
861 const Quaternion<T> twist = twist_quaternion_about_axis(q, unit_axis);
862 const T unsigned_angle = T(2) * std::atan2(twist.vec().length(), std::abs(twist.w()));
863 const T sign = twist.vec().dot(unit_axis) >= T(0) ? T(1) : T(-1);
864 return wrap_pi(sign * unsigned_angle);
865}

892 const Vec<T, 3> predicted_mag_world = tilt_orientation.rotate(normalized_or(mag_body, world_mag));
893 const Vec<T, 3> predicted_heading = project_onto_plane(predicted_mag_world, world_up);
894 const Vec<T, 3> reference_heading =
895 project_onto_plane(normalized_or(world_mag, Vec<T, 3>(T(1), T(0), T(0))), world_up);
896 if (!(predicted_heading.length_squared() > std::numeric_limits<T>::epsilon()) ||
897 !(reference_heading.length_squared() > std::numeric_limits<T>::epsilon()))
898 {
899 return tilt_orientation;
900 }
901
902 const T delta = normalized_or(predicted_heading, reference_heading)
903 .signed_angle(normalized_or(reference_heading, predicted_heading), world_up);
904 return (Quaternion<T>::from_axis_angle(world_up, delta) * tilt_orientation).normalized();
905}

1011template<typename T>
1012inline auto confidence_from_norm(T value, T expected, T min_value, T max_value) noexcept -> T
1013{
1014 if (!(value >= min_value && value <= max_value))
1015 {
1016 return T(0);
1017 }

1020 if (!(upper_span > std::numeric_limits<T>::epsilon()))

1022 return T(1);
1023 }
1024

2058 auto apply_vector_measurement_update(const Vec<T, 3>& measured_body, const Vec<T, 3>& reference_world, T variance,
2059 T nis_gate) noexcept -> bool
2060 {
2061 const Vec<T, 3> predicted_body = orientation_.rotate_inverse(reference_world);
2062 const Vec<T, 3> residual = measured_body - predicted_body;
2063
2064 Mat<T, 3, 6> h = Mat<T, 3, 6>::zero();
2065 const Mat<T, 3, 3> predicted_hat = detail::skew(predicted_body);
2066 for (std::size_t row = 0; row < 3; ++row)
2067 {
2068 for (std::size_t col = 0; col < 3; ++col)
2069 {
2070 h(row, col) = predicted_hat(row, col);
2071 }
2072 }
2073
2074 const Mat<T, 3, 3> r = detail::diagonal_covariance(variance);
2075 const Mat<T, 3, 3> s = h * covariance_ * h.transpose() + r;
2076 Mat<T, 3, 3> s_inv = Mat<T, 3, 3>::zero();
2077 if (!s.inverse(s_inv))
2078 {
2079 return false;
2080 }
2081
2082 const Vec<T, 3> s_inv_residual = s_inv * residual;
2083 const T nis = residual.dot(s_inv_residual);
2084 if (nis > nis_gate)
2085 {
2086 return false;
2087 }
2088
2089 const Mat<T, 6, 3> k = covariance_ * h.transpose() * s_inv;
2090 const Vec<T, 6> delta = k * residual;
2091 const Vec<T, 3> delta_theta(delta[0], delta[1], delta[2]);
2092 const Vec<T, 3> delta_bias(delta[3], delta[4], delta[5]);
2093 orientation_ = (orientation_ * detail::small_angle_quaternion(delta_theta)).normalized();
2094 gyro_bias_estimate_ = gyro_bias_estimate_ + delta_bias;
2095
2096 const Mat<T, 6, 6> i = Mat<T, 6, 6>::identity();
2097 const Mat<T, 6, 6> ikh = i - k * h;
2098 covariance_ = ikh * covariance_ * ikh.transpose() + k * r * k.transpose();
2099 return true;
2100 }
