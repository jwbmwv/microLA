469 [[nodiscard]] auto determinant() const -> T
470 {
471 static_assert(R == C, "Determinant is only defined for square matrices.");
472
473 if constexpr (R == 1)
474 {
475 return data[0];
476 }
477 else if constexpr (R == 2)
478 {
479 return data[0] * data[3] - data[1] * data[2];
480 }
481 else if constexpr (R == 3)
482 {
483 return data[0] * (data[4] * data[8] - data[5] * data[7]) -
484 data[1] * (data[3] * data[8] - data[5] * data[6]) +
485 data[2] * (data[3] * data[7] - data[4] * data[6]);
486 }
487 else
488 {
489 // Generic determinant using LU decomposition
490 // det(A) = det(P) * det(L) * det(U) = det(P) * det(U)
491 // det(L) = 1 (unit lower triangular), det(U) = product of diagonal
492 Mat temp = *this;
493 T det = T(1);
494
495 for (std::size_t i = 0; i < R; ++i)
496 {
497 // Find pivot
498 std::size_t pivot = i;
499 T max_val = std::abs(temp(i, i));
500 for (std::size_t k = i + 1; k < R; ++k)
501 {
502 T abs_val = std::abs(temp(k, i));
503 if (abs_val > max_val)
504 {
505 max_val = abs_val;
506 pivot = k;
507 }
508 }
509
510 if (max_val == T(0))
511 {
512 return T(0); // Singular matrix
513 }
514
515 // Swap rows if needed
516 if (pivot != i)
517 {
518 for (std::size_t j = 0; j < C; ++j)
519 {
520 std::swap(temp(i, j), temp(pivot, j));
521 }
522 det = -det; // Row swap changes sign
523 }
524
525 // Multiply determinant by pivot
526 det *= temp(i, i);
527
528 // Eliminate column below pivot
529 for (std::size_t k = i + 1; k < R; ++k)
530 {
531 T factor = temp(k, i) / temp(i, i);
532 for (std::size_t j = i; j < C; ++j)
533 {
534 temp(k, j) -= factor * temp(i, j);
535 }
536 }
537 }
538
539 return det;
540 }
541 }

549 // Non-throwing inverse: writes result into `out` and returns true on success.
550 [[nodiscard]] auto inverse(Mat& out) const noexcept -> bool
551 {
552 static_assert(R == C, "Inverse is only defined for square matrices.");
553 // Warn if matrix size may cause stack overflow
554 static_assert(R * C * sizeof(T) < 16384,
555 "Large matrix inverse may overflow embedded stack - consider dynamic allocation");
556
557 // Fast paths for small matrices
558 if constexpr (R == 2)
559 {
560 T det = data[0] * data[3] - data[1] * data[2];
561 if (det == T(0)) [[unlikely]]
562 {
563 return false;
564 }
565 out(0, 0) = data[3] / det;
566 out(0, 1) = -data[1] / det;
567 out(1, 0) = -data[2] / det;
568 out(1, 1) = data[0] / det;
569 return true;
570 }
571 else if constexpr (R == 3)
572 {
573 T det = data[0] * (data[4] * data[8] - data[5] * data[7]) -
574 data[1] * (data[3] * data[8] - data[5] * data[6]) +
575 data[2] * (data[3] * data[7] - data[4] * data[6]);
576 if (det == T(0)) [[unlikely]]
577 {
578 return false;
579 }
580 // Compute adjugate and divide by determinant
581 out(0, 0) = (data[4] * data[8] - data[5] * data[7]) / det;
582 out(0, 1) = (data[2] * data[7] - data[1] * data[8]) / det;
583 out(0, 2) = (data[1] * data[5] - data[2] * data[4]) / det;
584 out(1, 0) = (data[5] * data[6] - data[3] * data[8]) / det;
585 out(1, 1) = (data[0] * data[8] - data[2] * data[6]) / det;
586 out(1, 2) = (data[2] * data[3] - data[0] * data[5]) / det;
587 out(2, 0) = (data[3] * data[7] - data[4] * data[6]) / det;
588 out(2, 1) = (data[1] * data[6] - data[0] * data[7]) / det;
589 out(2, 2) = (data[0] * data[4] - data[1] * data[3]) / det;
590 return true;
591 }
592 else if constexpr (R == 4)
593 {
594 // Analytical inverse for 4x4 matrices using cofactor expansion
595 // This is 2-3× faster than Gauss-Jordan elimination (~200 cycles saved)
596 // Uses adjugate method: inv(A) = adj(A) / det(A)
597
598 // Compute 2×2 subdeterminants (used multiple times in cofactor calculation)
599 T s0 = data[0] * data[5] - data[4] * data[1];
600 T s1 = data[0] * data[6] - data[4] * data[2];
601 T s2 = data[0] * data[7] - data[4] * data[3];
602 T s3 = data[1] * data[6] - data[5] * data[2];
603 T s4 = data[1] * data[7] - data[5] * data[3];
604 T s5 = data[2] * data[7] - data[6] * data[3];
605
606 T c0 = data[10] * data[15] - data[14] * data[11];
607 T c1 = data[9] * data[15] - data[13] * data[11];
608 T c2 = data[9] * data[14] - data[13] * data[10];
609 T c3 = data[8] * data[15] - data[12] * data[11];
610 T c4 = data[8] * data[14] - data[12] * data[10];
611 T c5 = data[8] * data[13] - data[12] * data[9];
612
613 // Compute determinant using cofactor expansion
614 T det = s0 * c0 - s1 * c1 + s2 * c2 + s3 * c3 - s4 * c4 + s5 * c5;
615
616 if (det == T(0)) [[unlikely]]
617 {
618 return false;
619 }
620
621 T inv_det = T(1) / det;
622
623 // Compute adjugate matrix (transpose of cofactor matrix) and scale by 1/det
624 out(0, 0) = (data[5] * c0 - data[6] * c1 + data[7] * c2) * inv_det;
625 out(0, 1) = (-data[1] * c0 + data[2] * c1 - data[3] * c2) * inv_det;
626 out(0, 2) = (data[13] * s5 - data[14] * s4 + data[15] * s3) * inv_det;
627 out(0, 3) = (-data[9] * s5 + data[10] * s4 - data[11] * s3) * inv_det;
628
629 out(1, 0) = (-data[4] * c0 + data[6] * c3 - data[7] * c4) * inv_det;
630 out(1, 1) = (data[0] * c0 - data[2] * c3 + data[3] * c4) * inv_det;
631 out(1, 2) = (-data[12] * s5 + data[14] * s2 - data[15] * s1) * inv_det;
632 out(1, 3) = (data[8] * s5 - data[10] * s2 + data[11] * s1) * inv_det;
633
634 out(2, 0) = (data[4] * c1 - data[5] * c3 + data[7] * c5) * inv_det;
635 out(2, 1) = (-data[0] * c1 + data[1] * c3 - data[3] * c5) * inv_det;
636 out(2, 2) = (data[12] * s4 - data[13] * s2 + data[15] * s0) * inv_det;
637 out(2, 3) = (-data[8] * s4 + data[9] * s2 - data[11] * s0) * inv_det;
638
639 out(3, 0) = (-data[4] * c2 + data[5] * c4 - data[6] * c5) * inv_det;
640 out(3, 1) = (data[0] * c2 - data[1] * c4 + data[2] * c5) * inv_det;
641 out(3, 2) = (-data[12] * s3 + data[13] * s1 - data[14] * s0) * inv_det;
642 out(3, 3) = (data[8] * s3 - data[9] * s1 + data[10] * s0) * inv_det;
643
644 return true;
645 }
646 else
647 {
648 // Generic Gauss-Jordan elimination for larger matrices
649 // Create augmented matrix [A|I]
650 T augmented[R][2 * C];
651 for (std::size_t i = 0; i < R; ++i)
652 {
653 for (std::size_t j = 0; j < C; ++j)
654 {
655 augmented[i][j] = data[i * C + j];
656 augmented[i][j + C] = (i == j) ? T(1) : T(0);
657 }
658 }
659
660 // Forward elimination with partial pivoting
661 for (std::size_t i = 0; i < R; ++i)
662 {
663 // Find pivot
664 std::size_t pivot = i;
665 T max_val = std::abs(augmented[i][i]);
666 for (std::size_t k = i + 1; k < R; ++k)
667 {
668 T abs_val = std::abs(augmented[k][i]);
669 if (abs_val > max_val)
670 {
671 max_val = abs_val;
672 pivot = k;
673 }
674 }
675
676 if (max_val == T(0))
677 {
678 return false;
679 }
680
681 // Swap rows if needed
682 if (pivot != i)
683 {
684 for (std::size_t j = 0; j < 2 * C; ++j)
685 {
686 std::swap(augmented[i][j], augmented[pivot][j]);
687 }
688 }
689
690 // Scale pivot row
691 T pivot_val = augmented[i][i];
692 for (std::size_t j = 0; j < 2 * C; ++j)
693 {
694 augmented[i][j] /= pivot_val;
695 }
696
697 // Eliminate column
698 for (std::size_t k = 0; k < R; ++k)
699 {
700 if (k != i)
701 {
702 T factor = augmented[k][i];
703 for (std::size_t j = 0; j < 2 * C; ++j)
704 {
705 augmented[k][j] -= factor * augmented[i][j];
706 }
707 }
708 }
709 }
710
711 // Extract result from right half of augmented matrix
712 for (std::size_t i = 0; i < R; ++i)
713 {
714 for (std::size_t j = 0; j < C; ++j)
715 {
716 out(i, j) = augmented[i][j + C];
717 }
718 }
719 return true;
720 }
721 }

759 [[nodiscard]] auto inverse_checked(T threshold = T(1e6)) const noexcept -> std::variant<Mat, InverseError>
760 {
761 static_assert(R == C, "Inverse is only defined for square matrices.");
762
763 // Quick determinant check first
764 T det = determinant();
765 if (std::abs(det) < std::numeric_limits<T>::epsilon())
766 {
767 return InverseError::Singular;
768 }
769
770 // Compute inverse
771 auto inv_mat = try_inverse();
772 if (!inv_mat.has_value())
773 {
774 return InverseError::Singular;
775 }
776
777 // Check condition number: κ(A) = ||A|| * ||A^-1||
778 T norm_a = norm_inf();
779 T norm_a_inv = inv_mat->norm_inf();
780 T cond = norm_a * norm_a_inv;
781
782 if (cond > threshold)
783 {
784 return InverseError::IllConditioned;
785 }
786
787 return *inv_mat;
788 }

798 [[nodiscard]] auto pseudoinverse(T tolerance = 1e-9) const -> Mat<T, C, R>
799 {
800 // Compute SVD: A = U * S * V^T
801 auto [U, S, V] = svd();
802
803 // Compute S+ (pseudoinverse of S, which is R×C)
804 Mat<T, C, R> s_plus; // Note: dimensions are transposed
805 for (std::size_t i = 0; i < std::min(R, C); ++i)
806 {
807 T sigma = S(i, i);
808 if (std::abs(sigma) > tolerance)
809 {
810 s_plus(i, i) = static_cast<T>(1) / sigma;
811 }
812 else
813 {
814 s_plus(i, i) = static_cast<T>(0);
815 }
816 }
817
818 // Compute A+ = V * S+ * U^T
819 // First compute S+ * U^T
820 Mat<T, C, R> temp;
821 for (std::size_t i = 0; i < C; ++i)
822 {
823 for (std::size_t j = 0; j < R; ++j)
824 {
825 T sum = static_cast<T>(0);
826 for (std::size_t k = 0; k < std::min(C, R); ++k)
827 {
828 sum += s_plus(i, k) * U(j, k); // U^T means we access U(j, k) not U(k, j)
829 }
830 temp(i, j) = sum;
831 }
832 }
833
834 // Now compute V * temp
835 Mat<T, C, R> result;
836 for (std::size_t i = 0; i < C; ++i)
837 {

1082 // operating on contiguous rows/columns using 4-wide kernels. This
1083 // processes the k dimension in blocks of 4 and accumulates using

1156 {

1165 }
1166 }

1182 }
1183 }

2333 {

2339 {

2341 }
2342 T beta = T(2) / v_norm_sq;
2343

2345 {

2354 }
2355 }
2356

2358 {
2359 T sum = T(0);
2360 for (std::size_t j = k; j < R; ++j)
2361 {
2362 sum += q_out(i, j) * v[j];
2363 }
2364 for (std::size_t j = k; j < R; ++j)
2365 {
2366 q_out(i, j) -= beta * sum * v[j];
2367 }
2368 }
2369 }
