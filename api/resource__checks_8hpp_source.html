28template<typename T, std::size_t N>
29struct VecLayoutCheck
30{
31 using VecType = Vec<T, N>;
32
33 // Vec must be standard layout for DMA and C compatibility
34 static_assert(std::is_standard_layout_v<VecType>, "Vec must be standard layout for DMA compatibility");
35
36 // Vec should have minimal padding (size is either exact or rounded up to the active data alignment)
37 static_assert(sizeof(VecType) == sizeof(T) * N ||
38 sizeof(VecType) == ((sizeof(T) * N + MICROLA_DATA_ALIGNMENT - 1) &
39 ~static_cast<std::size_t>(MICROLA_DATA_ALIGNMENT - 1)),
40 "Vec has unexpected padding - check alignment settings");
41
42#ifdef MICROLA_STACK_SIZE_LIMIT
43 static_assert(sizeof(VecType) <= MICROLA_STACK_SIZE_LIMIT,
44 "Vec size exceeds MICROLA_STACK_SIZE_LIMIT - consider reducing dimension");
45#endif
46};

50template<typename T, std::size_t R, std::size_t C>
51struct MatLayoutCheck
52{
53 using MatType = Mat<T, R, C>;
54
55 static_assert(R > 0 && C > 0, "Matrix dimensions must be greater than 0");
56
57 // Mat must be standard layout for DMA and C compatibility
58 static_assert(std::is_standard_layout_v<MatType>, "Mat must be standard layout for DMA compatibility");
59
60#ifdef MICROLA_STACK_SIZE_LIMIT
61 static_assert(sizeof(T) * R * C <= MICROLA_STACK_SIZE_LIMIT,
62 "Matrix size exceeds MICROLA_STACK_SIZE_LIMIT - consider reducing dimensions");
63#endif
64
65 // Verify contiguous row-major storage
66 static_assert(sizeof(MatType) >= sizeof(T) * R * C, "Matrix storage is too small for elements");
67};

71template<typename T>
72struct QuaternionLayoutCheck
73{
74 using QuatType = Quaternion<T>;
75
76 static_assert(std::is_floating_point_v<T>, "Quaternion requires floating-point type");
77
78 // Quaternion must be standard layout for DMA and C compatibility
79 static_assert(std::is_standard_layout_v<QuatType>, "Quaternion must be standard layout for DMA compatibility");
80
81#ifdef MICROLA_STACK_SIZE_LIMIT
82 static_assert(sizeof(QuatType) <= MICROLA_STACK_SIZE_LIMIT, "Quaternion size exceeds MICROLA_STACK_SIZE_LIMIT");
83#endif
84
85 // Verify expected size (4 components + alignment padding up to MICROLA_DATA_ALIGNMENT)
86 static_assert(sizeof(QuatType) == sizeof(T) * 4 ||
87 sizeof(QuatType) == ((sizeof(T) * 4 + MICROLA_DATA_ALIGNMENT - 1) &
88 ~static_cast<std::size_t>(MICROLA_DATA_ALIGNMENT - 1)),
89 "Quaternion has unexpected size");
90};

112{

149template<typename T, std::size_t N>
150struct VecDMACheck
151{
152 // Verify Vec is trivially copyable (required for DMA)
153 static_assert(std::is_trivially_copyable_v<Vec<T, N>>, "Vec must be trivially copyable for DMA compatibility");
154
155 // Verify proper alignment for DMA (typically 16 bytes for SIMD)
156 static_assert(alignof(Vec<T, N>) >= alignof(T), "Vec alignment must be at least as strict as element type");
157};
