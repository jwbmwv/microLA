651 constexpr auto operator==(const Vec& other) const noexcept -> bool
652 {
653 // Optimized path for integral types (exact comparison)
654 if constexpr (std::is_integral_v<T>)
655 {
656 for (std::size_t i = 0; i < N; ++i)
657 {
658 if (data[i] != other.data[i])
659 {
660 return false;
661 }
662 }
663 return true;
664 }
665 // Floating-point types use epsilon comparison
666 else if constexpr (std::is_floating_point_v<T>)
667 {
668#ifdef CONFIG_MICROLA_NEON
669 // SIMD path for float equality comparison with epsilon
670 if constexpr (std::is_same_v<T, float> && N == 4)
671 {
672 float32x4_t a = vld1q_f32(data);
673 float32x4_t b = vld1q_f32(other.data);
674 float32x4_t diff = vabdq_f32(a, b);
675 float32x4_t eps = vdupq_n_f32(std::numeric_limits<float>::epsilon());
676 uint32x4_t cmp = vcleq_f32(diff, eps);
677 // All lanes must be true
678 uint64x2_t cmp64 = vreinterpretq_u64_u32(cmp);
679 return vgetq_lane_u64(cmp64, 0) == ~0ULL && vgetq_lane_u64(cmp64, 1) == ~0ULL;
680 }
681 if constexpr (std::is_same_v<T, float> && N == 3)
682 {
683 // Use approximate comparison for floats
684 for (std::size_t i = 0; i < 3; ++i)
685 {
686 if (std::abs(data[i] - other.data[i]) > std::numeric_limits<float>::epsilon())
687 return false;
688 }
689 return true;
690 }
691 if constexpr (std::is_same_v<T, float> && N == 2)
692 {
693 float32x2_t a = vld1_f32(data);
694 float32x2_t b = vld1_f32(other.data);
695 float32x2_t diff = vabd_f32(a, b);
696 float32x2_t eps = vdup_n_f32(std::numeric_limits<float>::epsilon());
697 uint32x2_t cmp = vcle_f32(diff, eps);
698 // Both lanes must be true
699 return vget_lane_u32(cmp, 0) == ~0U && vget_lane_u32(cmp, 1) == ~0U;
700 }
701#endif
702 // Fallback: epsilon comparison for all floating-point types
703 for (std::size_t i = 0; i < N; ++i)
704 {
705 if (std::abs(data[i] - other.data[i]) > std::numeric_limits<T>::epsilon())
706 {
707 return false;
708 }
709 }
710 return true;
711 }
712 // Fallback for other types: exact comparison
713 for (std::size_t i = 0; i < N; ++i)
714 {
715 if (data[i] != other.data[i])
716 {
717 return false;
718 }
719 }
720 return true;
721 }
