91 void predict(T dt) noexcept
92 {
93 if (!f || !F_jacobian || !std::isfinite(dt) || !x.is_finite() || !P.is_finite() || !Q.is_finite())
94 {
95 return;
96 }
97
98 const StateVec predicted_state = f(x, dt);
99 const StateMat jacobian = F_jacobian(x, dt);
100 if (!predicted_state.is_finite() || !jacobian.is_finite())
101 {
102 return;
103 }
104 const StateMat predicted_covariance = jacobian * P * jacobian.transpose() + Q;
105 if (predicted_covariance.is_finite())
106 {
107 x = predicted_state;
108 P = predicted_covariance;
109 }
110 }
111
125 auto update(const MeasVec& z) noexcept -> bool
126 {
127 if (!h || !H_jacobian || !z.is_finite() || !x.is_finite() || !P.is_finite() || !R.is_finite())
128 {
129 return false;
130 }
131
132 // Predict measurement from current state estimate
133 MeasVec z_pred = h(x);
134
135 // Compute measurement Jacobian at current state
136 MeasToStateMat h = H_jacobian(x);
137
138 // Innovation (measurement residual)
139 MeasVec y = residual ? residual(z, z_pred) : z - z_pred;
140 if (!y.is_finite() || !h.is_finite())
141 {
142 return false;
143 }
144
145 // Innovation covariance
146 MeasMat s = h * P * h.transpose() + R;
147
148 // Kalman gain
149 MeasMat s_inv;
150 if (!s.inverse_checked(s_inv))
151 {
152 return false;
153 }
154 StateToMeasMat k = P * h.transpose() * s_inv;
155
156 // Update state estimate
157 const StateVec next_state = x + k * y;
158
159 // Update covariance (simplified form)
160 StateMat i = StateMat::identity();
161 const StateMat next_covariance = (i - k * h) * P;
162 if (!next_state.is_finite() || !next_covariance.is_finite())
163 {
164 return false;
165 }
166 x = next_state;
167 P = next_covariance;
168
169 return true;
170 }
171
177 auto update_joseph(const MeasVec& z) noexcept -> bool
178 {
179 if (!h || !H_jacobian || !z.is_finite() || !x.is_finite() || !P.is_finite() || !R.is_finite())
180 {
181 return false;
182 }
183
184 MeasVec z_pred = h(x);
185 MeasToStateMat h = H_jacobian(x);
186 MeasVec y = residual ? residual(z, z_pred) : z - z_pred;
187 if (!y.is_finite() || !h.is_finite())
188 {
189 return false;
190 }
191 MeasMat s = h * P * h.transpose() + R;
192
193 MeasMat s_inv;
194 if (!s.inverse_checked(s_inv))
195 {
196 return false;
197 }
198 StateToMeasMat k = P * h.transpose() * s_inv;
199 const StateVec next_state = x + k * y;
200
201 // Joseph form covariance update
202 StateMat i = StateMat::identity();
203 StateMat ikh = i - k * h;
204 const StateMat next_covariance = ikh * P * ikh.transpose() + k * R * k.transpose();
205 if (!next_state.is_finite() || !next_covariance.is_finite())
206 {
207 return false;
208 }
209 x = next_state;
210 P = next_covariance;
211
212 return true;
213 }

294 [[nodiscard]] auto compute_innovation(const MeasVec& z) const noexcept -> MeasVec
295 {
296 if (!h)
297 {
298 return MeasVec();
299 }
300 const MeasVec predicted = h(x);
301 return residual ? residual(z, predicted) : z - predicted;
302 }

307 [[nodiscard]] auto compute_nis(const MeasVec& z) const noexcept -> T
308 {
309 if (!h || !H_jacobian || !z.is_finite() || !x.is_finite() || !P.is_finite() || !R.is_finite())
310 {
311 return std::numeric_limits<T>::infinity();
312 }
313
314 MeasVec y = compute_innovation(z);
315 MeasToStateMat h = H_jacobian(x);
316 MeasMat s = h * P * h.transpose() + R;
317 MeasMat s_inv;
318 if (!y.is_finite() || !s.inverse_checked(s_inv))
319 {
320 return std::numeric_limits<T>::infinity();
321 }
322
323 T nis = T(0);
324 for (std::size_t i = 0; i < MeasDim; ++i)
325 {
326 for (std::size_t j = 0; j < MeasDim; ++j)
327 {
328 nis += y[i] * s_inv(i, j) * y[j];
329 }
330 }
331 return std::isfinite(nis) && nis >= T(0) ? nis : std::numeric_limits<T>::infinity();
332 }
