26template<typename T = float, std::size_t StateDim = 2, std::size_t MeasDim = 1>
27class ExtendedKalmanFilter
28{
29public:
30 using StateVec = Vec<T, StateDim>;
31 using MeasVec = Vec<T, MeasDim>;
32 using StateMat = Mat<T, StateDim, StateDim>;
33 using MeasMat = Mat<T, MeasDim, MeasDim>;
34 using MeasToStateMat = Mat<T, MeasDim, StateDim>;
35 using StateToMeasMat = Mat<T, StateDim, MeasDim>;
36
38 using StateTransitionFunc = StateVec (*)(const StateVec&, T);
39
41 using MeasurementFunc = MeasVec (*)(const StateVec&);

44 using StateJacobianFunc = StateMat (*)(const StateVec&, T);
45

49private:

53 MeasMat R;

57 StateJacobianFunc F_jacobian;
58 MeasurementJacobianFunc H_jacobian;

60public:
68 ExtendedKalmanFilter(StateTransitionFunc state_func, MeasurementFunc meas_func, StateJacobianFunc state_jac,
69 MeasurementJacobianFunc meas_jac, const StateMat& process_noise,
70 const MeasMat& measurement_noise) noexcept

77 ExtendedKalmanFilter() noexcept
78 : x(), P(StateMat::identity()), Q(StateMat::identity() * T(0.01)), R(MeasMat::identity() * T(0.1)), f(nullptr),
79 h(nullptr), F_jacobian(nullptr), H_jacobian(nullptr)
80 {
81 }

89 void predict(T dt) noexcept
90 {
91 if (!f || !F_jacobian)
92 {
93 return;
94 }
95
96 // Propagate state through nonlinear dynamics
97 StateVec x_pred = f(x, dt);
98
99 // Compute Jacobian at current state
100 StateMat f = F_jacobian(x, dt);
101
102 // Update state
103 x = x_pred;
104
105 // Propagate covariance
106 P = f * P * f.transpose() + Q;
107 }
108
120 auto update(const MeasVec& z) noexcept -> bool
121 {
122 if (!h || !H_jacobian)
123 {
124 return false;
125 }
126
127 // Predict measurement from current state estimate
128 MeasVec z_pred = h(x);
129
130 // Compute measurement Jacobian at current state
131 MeasToStateMat h = H_jacobian(x);
132
133 // Innovation (measurement residual)
134 MeasVec y = z - z_pred;
135
136 // Innovation covariance
137 MeasMat s = h * P * h.transpose() + R;
138
139 // Check if innovation covariance is invertible
140 T det = s.determinant();
141 if (std::abs(det) < std::numeric_limits<T>::epsilon())
142 {
143 return false;
144 }
145
146 // Kalman gain
147 MeasMat s_inv;
148 if (!s.inverse(s_inv))
149 {
150 return false;
151 }
152 StateToMeasMat k = P * h.transpose() * s_inv;
153
154 // Update state estimate
155 x = x + k * y;
156
157 // Update covariance (simplified form)
158 StateMat i = StateMat::identity();
159 P = (i - k * h) * P;
160
161 return true;
162 }
163
167 auto update_joseph(const MeasVec& z) noexcept -> bool
168 {
169 if (!h || !H_jacobian)
170 {
171 return false;
172 }
173
174 MeasVec z_pred = h(x);
175 MeasToStateMat h = H_jacobian(x);
176 MeasVec y = z - z_pred;
177 MeasMat s = h * P * h.transpose() + R;
178
179 T det = s.determinant();
180 if (std::abs(det) < std::numeric_limits<T>::epsilon())
181 {
182 return false;
183 }
184
185 MeasMat s_inv;
186 if (!s.inverse(s_inv))
187 {
188 return false;
189 }
190 StateToMeasMat k = P * h.transpose() * s_inv;
191 x = x + k * y;
192
193 // Joseph form covariance update
194 StateMat i = StateMat::identity();
195 StateMat ikh = i - k * h;
196 P = ikh * P * ikh.transpose() + k * R * k.transpose();
197
198 return true;
199 }

244 {
245 if (!h)
246 {
247 return MeasVec();
248 }
249 return z - h(x);
250 }
251
253 [[nodiscard]] auto compute_nis(const MeasVec& z) const noexcept -> T
254 {
255 if (!h || !H_jacobian)
256 {
257 return T(0);
258 }
259
260 MeasVec y = z - h(x);
261 MeasToStateMat h = H_jacobian(x);
262 MeasMat s = h * P * h.transpose() + R;
263 MeasMat s_inv;
264 if (!s.inverse(s_inv))
265 {
266 return T(0);
267 }
268
269 T nis = T(0);
270 for (std::size_t i = 0; i < MeasDim; ++i)
271 {
272 for (std::size_t j = 0; j < MeasDim; ++j)
273 {
274 nis += y[i] * s_inv(i, j) * y[j];
275 }
276 }
277 return nis;
278 }
