40 // Imaginary part (x,y,z) first enables Vec<T,3> optimizations and better SIMD efficiency
41 // Memory layout: [x, y, z, w] where x,y,z are imaginary components (i,j,k) and w is real

70 Quaternion(const Vec<T, 3>& axis, T angle)
71 {
72 const T half_angle = angle * T(0.5);
73 const T s = std::sin(half_angle);
74 // Use Vec operations to compute imaginary part: xyz = axis * s
75 const Vec<T, 3> imaginary = axis * s;
76 data[X] = imaginary[X];
77 data[Y] = imaginary[Y];
78 data[Z] = imaginary[Z];
79 data[W] = std::cos(half_angle);
80 }

84 explicit Quaternion(const Mat<T, 3, 3>& mat)
85 {
86 const T trace = mat[0][0] + mat[1][1] + mat[2][2];
87 if (trace > T(0))
88 {
89 T s = std::sqrt(trace + T(1)) * T(2);
90 data[X] = (mat[1][2] - mat[2][1]) / s;
91 data[Y] = (mat[2][0] - mat[0][2]) / s;
92 data[Z] = (mat[0][1] - mat[1][0]) / s;
93 data[W] = s * T(0.25);
94 }
95 else if (mat[0][0] > mat[1][1] && mat[0][0] > mat[2][2])
96 {
97 T s = std::sqrt(T(1) + mat[0][0] - mat[1][1] - mat[2][2]) * T(2);
98 data[X] = s * T(0.25);
99 data[Y] = (mat[0][1] + mat[1][0]) / s;
100 data[Z] = (mat[2][0] + mat[0][2]) / s;
101 data[W] = (mat[1][2] - mat[2][1]) / s;
102 }
103 else if (mat[1][1] > mat[2][2])
104 {
105 T s = std::sqrt(T(1) + mat[1][1] - mat[0][0] - mat[2][2]) * T(2);
106 data[X] = (mat[0][1] + mat[1][0]) / s;
107 data[Y] = s * T(0.25);
108 data[Z] = (mat[1][2] + mat[2][1]) / s;
109 data[W] = (mat[2][0] - mat[0][2]) / s;
110 }
111 else
112 {
113 T s = std::sqrt(T(1) + mat[2][2] - mat[0][0] - mat[1][1]) * T(2);
114 data[X] = (mat[2][0] + mat[0][2]) / s;
115 data[Y] = (mat[1][2] + mat[2][1]) / s;
116 data[Z] = s * T(0.25);
117 data[W] = (mat[0][1] - mat[1][0]) / s;
118 }
119 }

376 [[nodiscard]] auto operator*(const Quaternion& other) const noexcept -> Quaternion
377 {
378#ifdef CONFIG_MICROLA_NEON
379 if constexpr (std::is_same_v<T, float>)
380 {
381 Quaternion result;
382 // Load quaternions: data layout is [x, y, z, w]
383 float32x4_t q1 = vld1q_f32(data); // [x1, y1, z1, w1]
384 float32x4_t q2 = vld1q_f32(other.data); // [x2, y2, z2, w2]
385
386 // Quaternion multiplication formulas:
387 // result.x = w1*x2 + x1*w2 + y1*z2 - z1*y2
388 // result.y = w1*y2 + y1*w2 + z1*x2 - x1*z2
389 // result.z = w1*z2 + z1*w2 + x1*y2 - y1*x2
390 // result.w = w1*w2 - x1*x2 - y1*y2 - z1*z2
391
392 // Broadcast each component of q1 (ARMv7 compatible)
393 float32x2_t q1_low = vget_low_f32(q1);
394 float32x2_t q1_high = vget_high_f32(q1);
395 float32x4_t q1_xxxx = vdupq_lane_f32(q1_low, 0); // [x1, x1, x1, x1]
396 float32x4_t q1_yyyy = vdupq_lane_f32(q1_low, 1); // [y1, y1, y1, y1]
397 float32x4_t q1_zzzz = vdupq_lane_f32(q1_high, 0); // [z1, z1, z1, z1]
398 float32x4_t q1_wwww = vdupq_lane_f32(q1_high, 1); // [w1, w1, w1, w1]
399
400 // Compute: w1*q2
401 float32x4_t t0 = vmulq_f32(q1_wwww, q2); // [w1*x2, w1*y2, w1*z2, w1*w2]
402
403 // Shuffle q2 for x1 terms: [w2, z2, y2, x2]
404 float32x4_t q2_wzyx = vrev64q_f32(q2); // [y2, x2, w2, z2]
405 q2_wzyx = vcombine_f32(vget_high_f32(q2_wzyx), vget_low_f32(q2_wzyx)); // [w2, z2, y2, x2]
406 float32x4_t t1 = vmulq_f32(q1_xxxx, q2_wzyx); // [x1*w2, x1*z2, x1*y2, x1*x2]
407
408 // Shuffle q2 for y1 terms: [z2, w2, x2, y2]
409 float32x4_t q2_zwxy = vextq_f32(q2, q2, 2); // [z2, w2, x2, y2]
410 float32x4_t t2 = vmulq_f32(q1_yyyy, q2_zwxy); // [y1*z2, y1*w2, y1*x2, y1*y2]
411
412 // Shuffle q2 for z1 terms: [y2, x2, w2, z2]
413 float32x4_t q2_yxwz = vrev64q_f32(q2); // [y2, x2, w2, z2]
414 float32x4_t t3 = vmulq_f32(q1_zzzz, q2_yxwz); // [z1*y2, z1*x2, z1*w2, z1*z2]
415
416 // Combine with correct signs: [+,+,+,-] for t1, [+,+,-,-] for t2, [-,+,+,-] for t3
417 float32x4_t sign1 = {1.0f, -1.0f, 1.0f, -1.0f};
418 float32x4_t sign2 = {1.0f, 1.0f, -1.0f, -1.0f};
419 float32x4_t sign3 = {-1.0f, 1.0f, 1.0f, -1.0f};
420
421 t1 = vmulq_f32(t1, sign1);
422 t2 = vmulq_f32(t2, sign2);
423 t3 = vmulq_f32(t3, sign3);
424
425 // Sum all terms
426 float32x4_t r = vaddq_f32(t0, t1);
427 r = vaddq_f32(r, t2);
428 r = vaddq_f32(r, t3);
429
430 vst1q_f32(result.data, r);
431 return result;
432 }
433#endif
434#ifdef CONFIG_MICROLA_CMSIS
435 if constexpr (std::is_same_v<T, float>)
436 {
437 Quaternion result;
438 arm_quaternion_product_f32(data, other.data, result.data);
439 return result;
440 }
441#endif
442 // Optimized scalar path using SIMD helpers for float
443 if constexpr (std::is_same_v<T, float>)
444 {
445 Quaternion result;
446 const float w1 = data[W];
447 const float x1 = data[X];
448 const float y1 = data[Y];
449 const float z1 = data[Z];
450 const float w2 = other.data[W];
451 const float x2 = other.data[X];
452 const float y2 = other.data[Y];
453 const float z2 = other.data[Z];
454
455 // Use simd helpers for 4-element operations
456 float t0[4] = {w1 * x2, w1 * y2, w1 * z2, w1 * w2};
457 float t1[4] = {x1 * w2, -x1 * z2, x1 * y2, -x1 * x2};
458 float t2[4] = {y1 * z2, y1 * w2, -y1 * x2, -y1 * y2};
459 const float t3[4] = {-z1 * y2, z1 * x2, z1 * w2, -z1 * z2};
460
461 microla::simd::add4(t0, t1, t0);
462 microla::simd::add4(t0, t2, t0);
463 microla::simd::add4(t0, t3, result.data);
464
465 return result;
466 }
467
468 return Quaternion((*this)[W] * other[W] - (*this)[X] * other[X] - (*this)[Y] * other[Y] - (*this)[Z] * other[Z],
469 (*this)[W] * other[X] + (*this)[X] * other[W] + (*this)[Y] * other[Z] - (*this)[Z] * other[Y],
470 (*this)[W] * other[Y] - (*this)[X] * other[Z] + (*this)[Y] * other[W] + (*this)[Z] * other[X],
471 (*this)[W] * other[Z] + (*this)[X] * other[Y] - (*this)[Y] * other[X] +
472 (*this)[Z] * other[W]);
473 }

478 [[nodiscard]] auto operator*(const Vec<T, 3>& v) const noexcept -> Vec<T, 3>
479 {
480 // Use the formula: v' = q * v * q^-1
481 // Optimized version: v' = v + 2 * cross(q.xyz, cross(q.xyz, v) + q.w * v)
482 const Vec<T, 3> qvec = vec();
483 const Vec<T, 3> uv = qvec.cross(v);
484 const Vec<T, 3> uuv = qvec.cross(uv);
485 return v + ((uv * (*this)[W]) + uuv) * T(2);
486 }

495 constexpr auto operator==(const Quaternion& other) const noexcept -> bool
496 {
497#ifdef CONFIG_MICROLA_NEON
498 if constexpr (std::is_same_v<T, float>)
499 {
500 float32x4_t a = vld1q_f32(data);
501 float32x4_t b = vld1q_f32(other.data);
502 float32x4_t diff = vabdq_f32(a, b);
503 float32x4_t eps = vdupq_n_f32(std::numeric_limits<float>::epsilon());
504 uint32x4_t cmp = vcleq_f32(diff, eps);
505 uint64x2_t cmp64 = vreinterpretq_u64_u32(cmp);
506 return vgetq_lane_u64(cmp64, 0) == ~0ULL && vgetq_lane_u64(cmp64, 1) == ~0ULL;
507 }
508#endif
509 // For floating point types, use epsilon comparison
510 if constexpr (std::is_floating_point<T>::value)
511 {
512 return std::abs((*this)[X] - other[X]) <= std::numeric_limits<T>::epsilon() &&
513 std::abs((*this)[Y] - other[Y]) <= std::numeric_limits<T>::epsilon() &&
514 std::abs((*this)[Z] - other[Z]) <= std::numeric_limits<T>::epsilon() &&
515 std::abs((*this)[W] - other[W]) <= std::numeric_limits<T>::epsilon();
516 }
517 // For integral types, use exact comparison
518 return (*this)[X] == other[X] && (*this)[Y] == other[Y] && (*this)[Z] == other[Z] && (*this)[W] == other[W];
519 }

564 MICROLA_CONSTEXPR_TRIG [[nodiscard]] auto norm() const noexcept -> T
565 {
566#ifdef CONFIG_MICROLA_NEON
567 if constexpr (std::is_same_v<T, float>)
568 {
569 float32x4_t a = vld1q_f32(data);
570 float32x4_t mul = vmulq_f32(a, a);
571 float32x2_t sum = vadd_f32(vget_low_f32(mul), vget_high_f32(mul));
572 sum = vpadd_f32(sum, sum);
573 return std::sqrt(static_cast<T>(vget_lane_f32(sum, 0)));
574 }
575#endif
576#ifdef CONFIG_MICROLA_CMSIS
577 if constexpr (std::is_same_v<T, float>)
578 {
579 float result;
580 arm_dot_prod_f32(data, data, 4, &result);
581 return std::sqrt(static_cast<T>(result));
582 }
583#endif
584 if constexpr (std::is_same_v<T, float>)
585 {
586 const auto* src = data;
587 float acc4[4] = {0.0F, 0.0F, 0.0F, 0.0F};
588 float tmp[4];
589 microla::simd::load_padded4(src, 4, tmp, 0.0F);
590 microla::simd::fma4_accumulate(acc4, tmp, tmp);
591 const float sum = acc4[0] + acc4[1] + acc4[2] + acc4[3];
592 return std::sqrt(static_cast<T>(sum));
593 }
594
595 // Use Vec<T,3> dot product for imaginary part + w^2
596 const Vec<T, 3> imag = vec();
597 return std::sqrt(imag.dot(imag) + (*this)[W] * (*this)[W]);
598 }

602 [[nodiscard]] auto normalized() const noexcept -> Quaternion
603 {
604#ifdef CONFIG_MICROLA_NEON
605 if constexpr (std::is_same_v<T, float>)
606 {
607 // Compute norm squared
608 float32x4_t a = vld1q_f32(data);
609 float32x4_t mul = vmulq_f32(a, a);
610 float32x2_t sum = vadd_f32(vget_low_f32(mul), vget_high_f32(mul));
611 sum = vpadd_f32(sum, sum);
612 float norm_sq = vget_lane_f32(sum, 0);
613
614 if (norm_sq != 0.0f)
615 {
616 // Fast inverse square root with Newton-Raphson refinement
617 float32x2_t norm_sq_v = vdup_n_f32(norm_sq);
618 float32x2_t rsqrt = vrsqrte_f32(norm_sq_v);
619 rsqrt = vmul_f32(rsqrt, vrsqrts_f32(vmul_f32(norm_sq_v, rsqrt), rsqrt));
620 rsqrt = vmul_f32(rsqrt, vrsqrts_f32(vmul_f32(norm_sq_v, rsqrt), rsqrt));
621
622 float inv_norm = vget_lane_f32(rsqrt, 0);

725 [[nodiscard]] static auto from_two_vectors(const Vec<T, 3>& from, const Vec<T, 3>& to) noexcept -> Quaternion
726 {
727 const Vec<T, 3> f = from.normalized();
728 const Vec<T, 3> t = to.normalized();
729 const T d = f.dot(t);
730
731 // Vectors are the same
732 if (d >= T(1) - T(1e-6))
733 {
734 return Quaternion(T(1), T(0), T(0), T(0));
735 }
736
737 // Vectors are opposite
738 if (d <= T(-1) + T(1e-6))
739 {
740 // Find an orthogonal axis
741 Vec<T, 3> axis = Vec<T, 3>(T(1), T(0), T(0)).cross(f);
742 if (axis.dot(axis) < T(1e-6))
743 {
744 axis = Vec<T, 3>(T(0), T(1), T(0)).cross(f);
745 }
746 return from_axis_angle(axis.normalized(), constants::pi<T>());
747 }
748
749 // Normal case
750 const Vec<T, 3> axis = f.cross(t);
751 const T s = std::sqrt((T(1) + d) * T(2));
752 const T inv_s = T(1) / s;
753
754 return Quaternion(s * T(0.5), axis[0] * inv_s, axis[1] * inv_s, axis[2] * inv_s);

798 [[nodiscard]] auto to_matrix() const noexcept -> Mat<T, 3, 3>
799 {
800 const T w = (*this)[W];
801 const T x = (*this)[X];
802 const T y = (*this)[Y];
803 const T z = (*this)[Z];
804 const T xx = x * x;
805 const T yy = y * y;
806 const T zz = z * z;
807 const T xy = x * y;
808 const T xz = x * z;
809 const T yz = y * z;
810 const T wx = w * x;
811 const T wy = w * y;
812 const T wz = w * z;

816 mat[0][2] = T(2) * (xz + wy);

1089 [[nodiscard]] auto slerp(const Quaternion& other, T t) const noexcept -> Quaternion
1090 {
1091 T cos_theta = dot(other);
1092
1093 // If quaternions are close, use linear interpolation
1094 if (std::abs(cos_theta) >= T(0.9995))
1095 {
1096 return lerp(other, t).normalized();
1097 }
1098
1099 // Handle negative dot product (choose shorter path)
1100 Quaternion q2 = other;
1101 if (cos_theta < T(0))
1102 {
1103 q2 = -other;
1104 cos_theta = -cos_theta;
1105 }
1106
1107 // Clamp to avoid numerical issues
1108 cos_theta = std::min(std::max(cos_theta, T(-1)), T(1));
1109
1110 const T theta = std::acos(cos_theta);
1111 const T sin_theta = std::sin(theta);
1112
1113 const T w1 = std::sin((T(1) - t) * theta) / sin_theta;
1114 const T w2 = std::sin(t * theta) / sin_theta;
1115
1116 return Quaternion((*this)[W] * w1 + q2[W] * w2, (*this)[X] * w1 + q2[X] * w2, (*this)[Y] * w1 + q2[Y] * w2,
1117 (*this)[Z] * w1 + q2[Z] * w2);

1149 {

1151

1154 const T cosr_cosp = T(1) - T(2) * ((*this)[X] * (*this)[X] + (*this)[Y] * (*this)[Y]);

1162 euler[PITCH] = std::copysign(constants::half_pi<T>(), sinp);

1165 {
