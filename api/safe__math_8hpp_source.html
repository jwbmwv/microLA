27template<typename T = float>
28inline auto is_finite(T value) noexcept -> bool
29{
30 static_assert(std::is_floating_point_v<T>, "is_finite requires floating-point type");
31 static_assert(sizeof(T) == 4 || sizeof(T) == 8, "is_finite supports 32-bit and 64-bit floating-point types");
32 using Bits = typename std::conditional<sizeof(T) == 4, std::uint32_t, std::uint64_t>::type;
33 Bits bits = 0;
34 std::memcpy(&bits, &value, sizeof(Bits));
35 if constexpr (sizeof(T) == 4)
36 {
37 constexpr std::uint32_t exp_mask32 = 0x7F800000U;
38 return (bits & static_cast<Bits>(exp_mask32)) != static_cast<Bits>(exp_mask32);
39 }
40 else if constexpr (sizeof(T) == 8)
41 {
42 constexpr std::uint64_t exp_mask64 = 0x7FF0000000000000ULL;
43 return (bits & static_cast<Bits>(exp_mask64)) != static_cast<Bits>(exp_mask64);
44 }
45
46 MICROLA_UNREACHABLE();
47 return false;
48}

54template<typename T = float>
55inline auto is_nan(T value) noexcept -> bool
56{
57 static_assert(std::is_floating_point_v<T>, "is_nan requires floating-point type");
58 static_assert(sizeof(T) == 4 || sizeof(T) == 8, "is_nan supports 32-bit and 64-bit floating-point types");
59 using Bits = typename std::conditional<sizeof(T) == 4, std::uint32_t, std::uint64_t>::type;
60 Bits bits = 0;
61 std::memcpy(&bits, &value, sizeof(Bits));
62 if constexpr (sizeof(T) == 4)
63 {
64 constexpr std::uint32_t exp_mask32 = 0x7F800000U;
65 constexpr std::uint32_t frac_mask32 = 0x007FFFFFU;
66 return (bits & static_cast<Bits>(exp_mask32)) == static_cast<Bits>(exp_mask32) &&
67 (bits & static_cast<Bits>(frac_mask32)) != static_cast<Bits>(0U);
68 }
69 else if constexpr (sizeof(T) == 8)
70 {
71 constexpr std::uint64_t exp_mask64 = 0x7FF0000000000000ULL;
72 constexpr std::uint64_t frac_mask64 = 0x000FFFFFFFFFFFFFULL;
73 return (bits & static_cast<Bits>(exp_mask64)) == static_cast<Bits>(exp_mask64) &&
74 (bits & static_cast<Bits>(frac_mask64)) != static_cast<Bits>(0ULL);
75 }

77 MICROLA_UNREACHABLE();
78 return false;

131template<typename T>
132inline auto safe_sqrt(T value) noexcept -> T
133{
134 static_assert(std::is_floating_point_v<T>, "safe_sqrt requires floating-point type");
135 // Preserve existing NaN to maintain diagnostic information
136 if (is_nan(value)) [[unlikely]]
137 {
138 return value;
139 }
140 // Return NaN for negative input to preserve mathematical correctness
141 if (value < T(0)) [[unlikely]]
142 {
143 return std::numeric_limits<T>::quiet_NaN();
144 }
145 return std::sqrt(value);
146}

249template<typename T>
250constexpr auto saturating_add(T a, T b) noexcept -> T
251{
252 static_assert(std::is_integral_v<T>, "saturating_add requires integral type");
253 if constexpr (std::is_signed_v<T>)
254 {
255 if (b > 0 && a > std::numeric_limits<T>::max() - b)
256 {
257 return std::numeric_limits<T>::max();
258 }
259 if (b < 0 && a < std::numeric_limits<T>::min() - b)
260 {
261 return std::numeric_limits<T>::min();
262 }
263 }
264 else
265 {
266 if (a > std::numeric_limits<T>::max() - b)
267 {
268 return std::numeric_limits<T>::max();
269 }
270 }
271 return a + b;
272}

311{
312 static_assert(std::is_integral_v<T>, "saturating_mul requires integral type");
313 if (a == 0 || b == 0)
