59template<typename T = float>
60inline auto sin(T x) noexcept -> T
61{
62 static_assert(std::is_floating_point_v<T>, "fast::sin requires floating-point type");
63
64 // Normalize to [-pi, pi]
65 const T pi = constants::pi<T>();
66 const T two_pi = constants::two_pi<T>();
67 x = x - std::floor((x + pi) / two_pi) * two_pi;
68
69 // Bhaskara I's sine approximation: sin(x) ≈ 16x(π - x) / (5π² - 4x(π - x))
70 if (x < T(0))
71 {
72 const T x_abs = -x;
73 const T numerator = T(16) * x_abs * (pi - x_abs);
74 const T denominator = T(5) * pi * pi - T(4) * x_abs * (pi - x_abs);
75 return -numerator / denominator;
76 }
77
78 const T numerator = T(16) * x * (pi - x);
79 const T denominator = T(5) * pi * pi - T(4) * x * (pi - x);
80 return numerator / denominator;
81}

99template<typename T = float>
100inline auto tan(T x) noexcept -> T
101{
102 const T sine = sin(x);
103 const T cosine = cos(x);
104 if (std::abs(cosine) < std::numeric_limits<T>::epsilon())
105 {
106 return sine >= T(0) ? std::numeric_limits<T>::infinity() : -std::numeric_limits<T>::infinity();
107 }
108 return sine / cosine;
109}

119template<int Iterations = 1>
120MICROLA_CONSTEXPR20 auto rsqrt(float x) noexcept -> float
121{
122 static_assert(Iterations >= 0 && Iterations <= 3, "Iterations must be 0-3");
123
124 // Quake III fast inverse square root
125 auto bits = MICROLA_BIT_CAST(std::uint32_t, x);
126 bits = RSQRT_MAGIC_CONSTANT - (bits >> 1);
127 auto y = MICROLA_BIT_CAST(float, bits);
128
129 // Newton-Raphson iterations for better accuracy
130 if constexpr (Iterations >= 1)
131 {
132 y = y * (1.5F - (0.5F * x * y * y));
133 }
134 if constexpr (Iterations >= 2)
135 {
136 y = y * (1.5F - (0.5F * x * y * y));
137 }
138 if constexpr (Iterations >= 3)
139 {
140 y = y * (1.5F - (0.5F * x * y * y));
141 }
142
143 return y;
144}

166template<typename T = float>
167inline auto atan2(T y, T x) noexcept -> T
168{
169 static_assert(std::is_floating_point_v<T>, "fast::atan2 requires floating-point type");
170
171 const T pi = constants::pi<T>();
172 const T half_pi = constants::half_pi<T>();
173
174 // Handle special cases
175 if (x == T(0))
176 {
177 if (y > T(0))
178 {
179 return half_pi;
180 }
181 if (y < T(0))
182 {
183 return -half_pi;
184 }
185 return T(0);
186 }
187
188 // Compute atan(y/x) using polynomial approximation
189 T z = y / x;
190 T abs_z = (z < T(0)) ? -z : z;
191
192 T angle;
193 if (abs_z < T(1))
194 {
195 // |z| < 1: use polynomial directly
196 angle = z / (T(1) + T(0.28) * z * z);
197 }
198 else
199 {
200 // |z| >= 1: use identity atan(z) = sign(z)*π/2 - atan(1/z)
201 T inv_z = T(1) / z;
202 angle = ((z > T(0)) ? half_pi : -half_pi) - inv_z / (T(1) + T(0.28) * inv_z * inv_z);
203 }
204
205 // Adjust for quadrant
206 if (x < T(0))
207 {
208 if (y >= T(0))
209 {
210 angle += pi;
211 }
212 else
213 {
214 angle -= pi;
215 }
216 }
217
218 return angle;
219}

225template<typename T = float>
226inline auto exp(T x) noexcept -> T
227{
228 static_assert(std::is_floating_point_v<T>, "fast::exp requires floating-point type");
229
230 // For large |x|, use scaling: e^x = (e^(x/2^k))^(2^k)
231 // Implement iteratively to avoid recursion (embedded-friendly).
232 int squares = 0;
233 while (x > T(2) || x < T(-2))
234 {
235 x = x / T(2);
236 ++squares;
237 }
238
239 // Padé approximation [2/2]: e^x ≈ (2 + x) / (2 - x)
240 // More accurate: e^x ≈ (12 + 6x + x²) / (12 - 6x + x²)
241 const T x2 = x * x;
242 const T numerator = T(12) + T(6) * x + x2;
243 const T denominator = T(12) - T(6) * x + x2;
244 T result = numerator / denominator;
245 for (int i = 0; i < squares; ++i)
246 {
247 result = result * result;
248 }
249 return result;
250}

256inline auto ln(float x) noexcept -> float
257{
258 // Extract exponent from IEEE 754 representation
259 auto bits = MICROLA_BIT_CAST(std::uint32_t, x);
260 const int exponent = static_cast<int>(((bits >> 23) & 0xFFU)) - 127;
261 bits = (bits & 0x007FFFFFU) | 0x3F800000U; // mantissa in [1, 2)
262 const auto m = MICROLA_BIT_CAST(float, bits);
263
264 // Polynomial approximation for ln(m) where m in [1, 2)
265 // ln(m) ≈ (m - 1) * (a + b*(m-1) + c*(m-1)²)
266 const float m1 = m - 1.0F;
267 const float log_m = m1 * (2.0F - 0.666666F * m1 + 0.4F * m1 * m1);
268
269 // ln(x) = ln(2^exp * m) = exp * ln(2) + ln(m)
270 return static_cast<float>(exponent) * constants::ln2<float>() + log_m;
271}

321template<typename T = float>
322inline auto smoothstep(T edge0, T edge1, T x) noexcept -> T
323{
324 T t = (x - edge0) / (edge1 - edge0);
325 if (t < T(0))
326 {
327 t = T(0);
328 }
329 if (t > T(1))
330 {
331 t = T(1);
332 }
333 return t * t * (T(3) - T(2) * t);
334}
