79 [[nodiscard]] static auto from_point_normal(const Vec<T, 3>& point, const Vec<T, 3>& normal) -> Plane
80 {
81 Vec<T, 3> n = normal.normalized();

90 [[nodiscard]] static auto from_points(const Vec<T, 3>& p0, const Vec<T, 3>& p1, const Vec<T, 3>& p2) -> Plane
91 {
92 Vec<T, 3> v1 = p1 - p0;
93 Vec<T, 3> v2 = p2 - p0;
94 Vec<T, 3> n = v1.cross(v2).normalized();
95 return Plane(n, -n.dot(p0));

102 [[nodiscard]] auto distance(const Vec<T, 3>& point) const -> T
103 {
104 // Use Kahan summation for numerical stability
105 T sum = normal.x() * point.x();
106 T c = T(0);
107
108 T y = normal.y() * point.y() - c;
109 T t = sum + y;
110 c = (t - sum) - y;
111 sum = t;
112
113 y = normal.z() * point.z() - c;
114 t = sum + y;
115 c = (t - sum) - y;
116 sum = t;
117
118 y = d - c;
119 t = sum + y;
120
121 return t;
122 }
123
125 [[nodiscard]] auto signed_distance(const Vec<T, 3>& point) const -> T { return distance(point); }

130 [[nodiscard]] auto project(const Vec<T, 3>& point) const -> Vec<T, 3>
131 {
132 // Compute projection components using a selectable higher-precision
133 // intermediate type to avoid unnecessary long-double bloat on embedded
134 // targets.
135 using high_t = microla::detail::hi_t<T>;
136
137 auto nx = static_cast<high_t>(normal.x());
138 auto ny = static_cast<high_t>(normal.y());
139 auto nz = static_cast<high_t>(normal.z());
140
141 auto px = static_cast<high_t>(point.x());
142 auto py = static_cast<high_t>(point.y());
143 auto pz = static_cast<high_t>(point.z());
144
145 auto dd = static_cast<high_t>(d);
146
147 high_t dist = nx * px + ny * py + nz * pz + dd;
148
149 T rx = static_cast<T>(px - nx * dist);
150 T ry = static_cast<T>(py - ny * dist);
151 T rz = static_cast<T>(pz - nz * dist);
152
153 return Vec<T, 3>(rx, ry, rz);
154 }
155
159 [[nodiscard]] auto intersect(const Ray<T>& ray) const -> std::optional<T>
160 {
161 T denom = normal.dot(ray.direction);
162
163 // Check if ray is parallel to plane
164 if (std::abs(denom) < std::numeric_limits<T>::epsilon())
165 {
166 return std::nullopt;
167 }
168
169 // For Hessian form n·x + d = 0, solve for t where n·(o + t*dir) + d = 0
170 T t = -(normal.dot(ray.origin) + d) / denom;
171
172 return (t >= static_cast<T>(0)) ? std::optional<T>(t) : std::nullopt;
173 }
174};

183template<typename T = float>
184struct AABB
185{
187 Vec<T, 3> min;
188
190 Vec<T, 3> max;
191
193 AABB() : min(Vec<T, 3>()), max(Vec<T, 3>()) {}
194
198 AABB(const Vec<T, 3>& minimum, const Vec<T, 3>& maximum) : min(minimum), max(maximum) {}
199
204 [[nodiscard]] static auto from_center_extents(const Vec<T, 3>& center, const Vec<T, 3>& extents) -> AABB
205 {
206 return AABB(center - extents, center + extents);
207 }
208
211 [[nodiscard]] auto center() const -> Vec<T, 3> { return (min + max) * static_cast<T>(0.5); }
212
215 [[nodiscard]] auto extents() const -> Vec<T, 3> { return (max - min) * static_cast<T>(0.5); }
216
219 [[nodiscard]] auto size() const -> Vec<T, 3> { return max - min; }
220
224 [[nodiscard]] auto contains(const Vec<T, 3>& point) const -> bool
225 {
226 return (point.x() >= min.x() && point.x() <= max.x()) && (point.y() >= min.y() && point.y() <= max.y()) &&
227 (point.z() >= min.z() && point.z() <= max.z());
228 }
229
233 [[nodiscard]] auto intersects(const AABB& other) const -> bool
234 {
235 return (min.x() <= other.max.x() && max.x() >= other.min.x()) &&
236 (min.y() <= other.max.y() && max.y() >= other.min.y()) &&
237 (min.z() <= other.max.z() && max.z() >= other.min.z());
238 }
239
243 [[nodiscard]] auto merge(const AABB& other) const -> AABB
244 {
245 return AABB(Vec<T, 3>(std::min(min.x(), other.min.x()), std::min(min.y(), other.min.y()),
246 std::min(min.z(), other.min.z())),
247 Vec<T, 3>(std::max(max.x(), other.max.x()), std::max(max.y(), other.max.y()),
248 std::max(max.z(), other.max.z())));
249 }
250
253 void expand(const Vec<T, 3>& point)
254 {
255 min = Vec<T, 3>(std::min(min.x(), point.x()), std::min(min.y(), point.y()), std::min(min.z(), point.z()));
256 max = Vec<T, 3>(std::max(max.x(), point.x()), std::max(max.y(), point.y()), std::max(max.z(), point.z()));
257 }
258
262 [[nodiscard]] auto expand(T amount) const -> AABB
263 {
264 Vec<T, 3> offset(amount, amount, amount);
265 return AABB(min - offset, max + offset);
266 }
267
271 [[nodiscard]] auto contains(const AABB& other) const -> bool
272 {
273 return (other.min.x() >= min.x() && other.max.x() <= max.x()) &&
274 (other.min.y() >= min.y() && other.max.y() <= max.y()) &&
275 (other.min.z() >= min.z() && other.max.z() <= max.z());
276 }
277
281 [[nodiscard]] auto intersect(const Ray<T>& ray) const -> std::optional<std::pair<T, T>>
282 {
283 T t_min = static_cast<T>(0);
284 T t_max = std::numeric_limits<T>::infinity();
285
286 for (int i = 0; i < 3; ++i)
287 {
288 // Check for axis-aligned ray (avoid division by zero)
289 if (std::abs(ray.direction[i]) < std::numeric_limits<T>::epsilon())
290 {
291 // Ray is parallel to the slab, check if it's within bounds
292 if (ray.origin[i] < min[i] || ray.origin[i] > max[i])
293 {
294 return std::nullopt; // Ray is outside the slab
295 }
296 continue; // Ray is within the slab, check other axes
297 }
298
299 T inv_d = static_cast<T>(1) / ray.direction[i];
300 T t0 = (min[i] - ray.origin[i]) * inv_d;
301 T t1 = (max[i] - ray.origin[i]) * inv_d;
302
303 if (inv_d < static_cast<T>(0))
304 {
305 std::swap(t0, t1);
306 }
307
308 t_min = std::max(t_min, t0);
309 t_max = std::min(t_max, t1);
310
311 if (t_max < t_min)
312 {
313 return std::nullopt;
314 }
315 }
316
317 return (t_min >= static_cast<T>(0)) ? std::optional<std::pair<T, T>>(std::make_pair(t_min, t_max))
318 : std::nullopt;
319 }
320
323 [[nodiscard]] auto surface_area() const -> T
324 {
325 Vec<T, 3> s = size();
326 return static_cast<T>(2) * (s.x() * s.y() + s.y() * s.z() + s.z() * s.x());
327 }
328
331 [[nodiscard]] auto volume() const -> T
332 {
333 Vec<T, 3> s = size();
334 return s.x() * s.y() * s.z();
335 }
336};

345template<typename T = float>
346struct Sphere
347{
349 Vec<T, 3> center;
350
352 T radius;
353
355 Sphere() : center(), radius(T(1)) {}
356
360 Sphere(const Vec<T, 3>& c, T r) : center(c), radius(r) {}
361
365 [[nodiscard]] auto contains(const Vec<T, 3>& point) const -> bool
366 {
367 Vec<T, 3> diff = point - center;
368 return diff.dot(diff) <= radius * radius;
369 }
370
374 [[nodiscard]] auto intersects(const Sphere& other) const -> bool
375 {
376 Vec<T, 3> diff = center - other.center;
377 T dist_sq = diff.dot(diff);
378 T radius_sum = radius + other.radius;
379 return dist_sq <= radius_sum * radius_sum;
380 }
381
384 [[nodiscard]] auto to_aabb() const -> AABB<T>
385 {
386 Vec<T, 3> extents(radius, radius, radius);
387 return AABB<T>(center - extents, center + extents);
388 }
389
392 [[nodiscard]] auto surface_area() const -> T { return static_cast<T>(4) * constants::pi<T>() * radius * radius; }
393
396 [[nodiscard]] auto volume() const -> T
397 {
398 return static_cast<T>(4.0 / 3.0) * constants::pi<T>() * radius * radius * radius;
399 }
400
404 [[nodiscard]] auto intersect(const Ray<T>& ray) const -> std::optional<std::pair<T, T>>
405 {
406 Vec<T, 3> oc = ray.origin - center;
407
408 T a = ray.direction.dot(ray.direction);
409 T b = static_cast<T>(2) * oc.dot(ray.direction);
410 T c = oc.dot(oc) - radius * radius;
411
412 T discriminant = b * b - static_cast<T>(4) * a * c;
413
414 if (discriminant < static_cast<T>(0))
415 {
416 return std::nullopt;
417 }
418
419 T sqrt_disc = std::sqrt(discriminant);
420 T t1 = (-b - sqrt_disc) / (static_cast<T>(2) * a);
421 T t2 = (-b + sqrt_disc) / (static_cast<T>(2) * a);
422
423 return std::make_pair(t1, t2);
424 }
425
429 [[nodiscard]] auto contains(const Sphere& other) const -> bool
430 {
431 Vec<T, 3> diff = center - other.center;
432 T dist = std::sqrt(diff.dot(diff));
433 return dist + other.radius <= radius;
434 }
435
439 [[nodiscard]] auto intersects(const AABB<T>& aabb) const -> bool
440 {
441 // Find closest point on AABB to sphere center
442 Vec<T, 3> closest(microla::clamp(center.x(), aabb.min.x(), aabb.max.x()),
443 microla::clamp(center.y(), aabb.min.y(), aabb.max.y()),
444 microla::clamp(center.z(), aabb.min.z(), aabb.max.z()));
445
446 Vec<T, 3> diff = closest - center;
447 return diff.dot(diff) <= radius * radius;
448 }
449};

461template<typename T>
462auto intersect(const Ray<T>& ray, const Sphere<T>& sphere) -> std::optional<T>
463{
464 Vec<T, 3> oc = ray.origin - sphere.center;
465
466 T a = ray.direction.dot(ray.direction);
467 T b = static_cast<T>(2) * oc.dot(ray.direction);
468 T c = oc.dot(oc) - sphere.radius * sphere.radius;
469
470 T discriminant = b * b - static_cast<T>(4) * a * c;
471
472 if (discriminant < static_cast<T>(0))
473 {
474 return std::nullopt; // No intersection
475 }
476
477 T sqrt_disc = std::sqrt(discriminant);
478 T t = (-b - sqrt_disc) / (static_cast<T>(2) * a);
479
480 if (t < static_cast<T>(0))
481 {
482 t = (-b + sqrt_disc) / (static_cast<T>(2) * a);
483 }
484
485 return (t >= static_cast<T>(0)) ? std::optional<T>(t) : std::nullopt;
486}

493template<typename T>
494auto intersect(const Ray<T>& ray, const Plane<T>& plane) -> std::optional<T>
495{
496 T denom = plane.normal.dot(ray.direction);
497
498 // Check if ray is parallel to plane
499 if (std::abs(denom) < std::numeric_limits<T>::epsilon())
500 {
501 return std::nullopt;
502 }
503
504 // For Hessian form n·x + d = 0, solve for t where n·(o + t*dir) + d = 0
505 // => n·o + t*(n·dir) + d = 0 => t = -(n·o + d) / (n·dir)
506 T t = -(plane.normal.dot(ray.origin) + plane.d) / denom;
507
508 return (t >= static_cast<T>(0)) ? std::optional<T>(t) : std::nullopt;
509}

566template<typename T>
567struct Triangle
568{
569 Vec<T, 3> v0, v1, v2;
570
572 Triangle() : v0(), v1(), v2() {}
573
578 Triangle(const Vec<T, 3>& a, const Vec<T, 3>& b, const Vec<T, 3>& c) : v0(a), v1(b), v2(c) {}
579
582 [[nodiscard]] auto normal() const -> Vec<T, 3>
583 {
584 Vec<T, 3> e1 = v1 - v0;
585 Vec<T, 3> e2 = v2 - v0;
586 return e1.cross(e2).normalized();
587 }
588
591 [[nodiscard]] auto area() const -> T
592 {
593 Vec<T, 3> e1 = v1 - v0;
594 Vec<T, 3> e2 = v2 - v0;
595 return e1.cross(e2).length() * static_cast<T>(0.5);
596 }
597
600 [[nodiscard]] auto centroid() const -> Vec<T, 3>
601 {
602 return (v0 + v1 + v2) * (static_cast<T>(1) / static_cast<T>(3));
603 }
604
607 [[nodiscard]] auto to_aabb() const -> AABB<T>
608 {
609 AABB<T> box;
610 box.expand(v0);
611 box.expand(v1);
612 box.expand(v2);
613 return box;
614 }
615
619 [[nodiscard]] auto contains(const Vec<T, 3>& point) const -> bool
620 {
621 // Barycentric coordinate method
622 Vec<T, 3> v0v1 = v1 - v0;
623 Vec<T, 3> v0v2 = v2 - v0;
624 Vec<T, 3> v0p = point - v0;
625
626 T dot00 = v0v1.dot(v0v1);
627 T dot01 = v0v1.dot(v0v2);
628 T dot02 = v0v1.dot(v0p);
629 T dot11 = v0v2.dot(v0v2);
630 T dot12 = v0v2.dot(v0p);
631
632 T inv_denom = static_cast<T>(1) / (dot00 * dot11 - dot01 * dot01);
633 T u = (dot11 * dot02 - dot01 * dot12) * inv_denom;
634 T v = (dot00 * dot12 - dot01 * dot02) * inv_denom;
635
636 return (u >= static_cast<T>(0)) && (v >= static_cast<T>(0)) && (u + v <= static_cast<T>(1));
637 }
638
642 [[nodiscard]] auto intersect(const Ray<T>& ray) const -> std::optional<T>
643 {
644 constexpr T epsilon = std::numeric_limits<T>::epsilon();
645
646 Vec<T, 3> edge1 = v1 - v0;
647 Vec<T, 3> edge2 = v2 - v0;
648
649 Vec<T, 3> h = ray.direction.cross(edge2);
650 T a = edge1.dot(h);
651
652 // Ray parallel to triangle
653 if (std::abs(a) < epsilon)
654 {
655 return std::nullopt;
656 }
657
658 T f = static_cast<T>(1) / a;
659 Vec<T, 3> s = ray.origin - v0;
660 T u = f * s.dot(h);
661
662 if (u < static_cast<T>(0) || u > static_cast<T>(1))
663 {
664 return std::nullopt;
665 }
666
667 Vec<T, 3> q = s.cross(edge1);
668 T v = f * ray.direction.dot(q);
669
670 if (v < static_cast<T>(0) || u + v > static_cast<T>(1))
671 {
672 return std::nullopt;
673 }
674
675 T t = f * edge2.dot(q);
676
677 return (t > epsilon) ? std::optional<T>(t) : std::nullopt;
678 }
679};

758 [[nodiscard]] static auto from_matrix(const Mat<T, 4, 4>& vp) -> Frustum

774

839 {
840 for (const auto& plane : planes)
841 {
842 // Find positive vertex (farthest along plane normal)
