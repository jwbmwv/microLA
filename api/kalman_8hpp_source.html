110 auto update(const MeasVec& z) noexcept -> bool
111 {
112 if (!z.is_finite() || !x.is_finite() || !P.is_finite() || !H.is_finite() || !R.is_finite())
113 {
114 return false;
115 }
116 // Innovation (measurement residual): y = z - H * x
117 MeasVec y = z - H * x;
118
119 // Innovation covariance: S = H * P * H^T + R
120 MeasMat s = H * P * H.transpose() + R;
121
122 // Kalman gain: K = P * H^T * S^(-1)
123 // Use condition-number-aware inversion: rejects ill-conditioned S regardless
124 // of the absolute scale of measurement units (absolute det < epsilon is insufficient).
125 MeasMat s_inv;
126 if (!s.inverse_checked(s_inv))
127 {
128 return false;
129 }
130 StateToMeasMat k = P * H.transpose() * s_inv;
131
132 // Update state estimate: x = x + K * y
133 const StateVec next_state = x + k * y;
134
135 // Update covariance using Joseph form for numerical stability
136 // P = (I - K*H) * P * (I - K*H)^T + K * R * K^T
137 StateMat i = StateMat::identity();
138 StateMat ikh = i - k * H;
139 StateMat next_covariance = ikh * P * ikh.transpose() + k * R * k.transpose();
140
141 // Enforce covariance symmetry
142 // Symmetrize: P = (P + P^T) / 2
143 next_covariance = (next_covariance + next_covariance.transpose()) * static_cast<T>(0.5);
144 if (!next_state.is_finite() || !next_covariance.is_finite())
145 {
146 return false;
147 }
148 x = next_state;
149 P = next_covariance;
150
151 return true;
152 }
153
161 auto update_simple(const MeasVec& z) noexcept -> bool
162 {
163 if (!z.is_finite() || !x.is_finite() || !P.is_finite() || !H.is_finite() || !R.is_finite())
164 {
165 return false;
166 }
167 MeasVec y = z - H * x;
168 MeasMat s = H * P * H.transpose() + R;
169
170 MeasMat s_inv;
171 if (!s.inverse_checked(s_inv))
172 {
173 return false;
174 }
175 StateToMeasMat k = P * H.transpose() * s_inv;
176 const StateVec next_state = x + k * y;
177
178 // Simplified covariance update
179 StateMat i = StateMat::identity();
180 const StateMat next_covariance = (i - k * H) * P;
181 if (!next_state.is_finite() || !next_covariance.is_finite())
182 {
183 return false;
184 }
185 x = next_state;
186 P = next_covariance;
187
188 return true;
189 }

310 [[nodiscard]] auto compute_nis(const MeasVec& z) const noexcept -> T
311 {
312 if (!z.is_finite() || !x.is_finite() || !P.is_finite() || !H.is_finite() || !R.is_finite())
313 {
314 return std::numeric_limits<T>::infinity();
315 }
316 MeasVec y = compute_innovation(z);
317 MeasMat s = compute_innovation_covariance();
318 MeasMat s_inv;
319 if (!s.inverse_checked(s_inv))
320 {
321 return std::numeric_limits<T>::infinity();
322 }
323
324 // NIS = y^T * S^(-1) * y
325 T nis = T(0);
326 for (std::size_t i = 0; i < MeasDim; ++i)
327 {
328 for (std::size_t j = 0; j < MeasDim; ++j)
329 {
330 nis += y[i] * s_inv(i, j) * y[j];
331 }
332 }
333 return std::isfinite(nis) && nis >= T(0) ? nis : std::numeric_limits<T>::infinity();
334 }
