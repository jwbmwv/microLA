95 auto update(const MeasVec& z) noexcept -> bool
96 {
97 // Innovation (measurement residual): y = z - H * x
98 MeasVec y = z - H * x;
99
100 // Innovation covariance: S = H * P * H^T + R
101 MeasMat s = H * P * H.transpose() + R;
102
103 // Kalman gain: K = P * H^T * S^(-1)
104 // Use condition-number-aware inversion: rejects ill-conditioned S regardless
105 // of the absolute scale of measurement units (absolute det < epsilon is insufficient).
106 MeasMat s_inv;
107#if !defined(MICROLA_EMBEDDED)
108 auto s_result = s.inverse_checked();
109 if (std::holds_alternative<typename MeasMat::InverseError>(s_result))
110 {
111 return false; // Singular or ill-conditioned innovation covariance
112 }
113 s_inv = std::get<MeasMat>(s_result);
114#else
115 // Embedded: std::variant not available; use lightweight bool form.
116 if (!s.inverse(s_inv))
117 {
118 return false; // Singular innovation covariance
119 }
120#endif
121 StateToMeasMat k = P * H.transpose() * s_inv;
122
123 // Update state estimate: x = x + K * y
124 x = x + k * y;
125
126 // Update covariance using Joseph form for numerical stability
127 // P = (I - K*H) * P * (I - K*H)^T + K * R * K^T
128 StateMat i = StateMat::identity();
129 StateMat ikh = i - k * H;
130 P = ikh * P * ikh.transpose() + k * R * k.transpose();
131
132 // Enforce covariance symmetry and positive definiteness
133 // Symmetrize: P = (P + P^T) / 2
134 P = (P + P.transpose()) * static_cast<T>(0.5);
135
136 return true;
137 }

143 auto update_simple(const MeasVec& z) noexcept -> bool
144 {
145 MeasVec y = z - H * x;
146 MeasMat s = H * P * H.transpose() + R;
147
148 MeasMat s_inv;
149#if !defined(MICROLA_EMBEDDED)
150 auto s_result = s.inverse_checked();
151 if (std::holds_alternative<typename MeasMat::InverseError>(s_result))
152 {
153 return false; // Singular or ill-conditioned innovation covariance
154 }
155 s_inv = std::get<MeasMat>(s_result);
156#else
157 if (!s.inverse(s_inv))
158 {
159 return false;
160 }
161#endif
162 StateToMeasMat k = P * H.transpose() * s_inv;
163 x = x + k * y;
164
165 // Simplified covariance update
166 StateMat i = StateMat::identity();
167 P = (i - k * H) * P;
168
169 return true;
170 }

266 [[nodiscard]] auto compute_nis(const MeasVec& z) const noexcept -> T
267 {
268 MeasVec y = compute_innovation(z);
269 MeasMat s = compute_innovation_covariance();
270 MeasMat s_inv;
271 if (!s.inverse(s_inv))
272 {
273 return T(0);
274 }
275
276 // NIS = y^T * S^(-1) * y
277 T nis = T(0);
278 for (std::size_t i = 0; i < MeasDim; ++i)
279 {
280 for (std::size_t j = 0; j < MeasDim; ++j)
281 {
282 nis += y[i] * s_inv(i, j) * y[j];
283 }
284 }
285 return nis;
286 }
